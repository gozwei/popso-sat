cmake_minimum_required(VERSION 3.20)
project(popso-sat LANGUAGES CXX VERSION 0.3.1)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)

if(NOT CMAKE_BUILD_TYPE AND NOT CMAKE_CONFIGURATION_TYPES)
  set(CMAKE_BUILD_TYPE Release CACHE STRING "Build type" FORCE)
endif()

option(PSAT_NATIVE "Optimize for the host CPU (-march=native)" OFF)
option(PSAT_SANITIZE "Build with address/undefined sanitizers" OFF)
option(PSAT_BUILD_TESTS "Build unit tests" ON)
option(PSAT_WERROR "Treat compiler warnings as errors (CI)" OFF)

find_package(Threads REQUIRED)

if(MSVC)
  set(PSAT_WARNINGS /W4 /permissive- /wd4100 /wd4244 /wd4267)
  if(PSAT_WERROR)
    list(APPEND PSAT_WARNINGS /WX)
  endif()
else()
  set(PSAT_WARNINGS -Wall -Wextra -Wpedantic -Wshadow -Wno-unused-parameter)
  if(PSAT_WERROR)
    list(APPEND PSAT_WARNINGS -Werror)
  endif()
endif()

file(GLOB_RECURSE PSAT_SOURCES CONFIGURE_DEPENDS ${CMAKE_SOURCE_DIR}/src/psat/*.cpp)
add_library(psat_lib STATIC ${PSAT_SOURCES})
target_include_directories(psat_lib PUBLIC ${CMAKE_SOURCE_DIR}/src)
target_compile_options(psat_lib PRIVATE ${PSAT_WARNINGS})
target_compile_definitions(psat_lib PUBLIC PSAT_VERSION="${PROJECT_VERSION}")
target_link_libraries(psat_lib PUBLIC Threads::Threads)
set_target_properties(psat_lib PROPERTIES POSITION_INDEPENDENT_CODE ON)
if(WIN32)
  target_compile_definitions(psat_lib PUBLIC _CRT_SECURE_NO_WARNINGS NOMINMAX WIN32_LEAN_AND_MEAN)
endif()
if(MSVC)
  target_compile_options(psat_lib PUBLIC /utf-8)
endif()

if(PSAT_NATIVE AND NOT MSVC)
  target_compile_options(psat_lib PUBLIC -march=native)
endif()
if(PSAT_SANITIZE AND NOT MSVC)
  target_compile_options(psat_lib PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_options(psat_lib PUBLIC -fsanitize=address,undefined)
endif()

add_executable(popso-sat apps/popso_sat_main.cpp)
target_link_libraries(popso-sat PRIVATE psat_lib)
target_compile_options(popso-sat PRIVATE ${PSAT_WARNINGS})

if(PSAT_BUILD_TESTS)
  enable_testing()
  file(GLOB PSAT_TEST_SOURCES CONFIGURE_DEPENDS ${CMAKE_SOURCE_DIR}/tests/*.cpp)
  add_executable(psat_tests ${PSAT_TEST_SOURCES})
  target_link_libraries(psat_tests PRIVATE psat_lib)
  target_compile_options(psat_tests PRIVATE ${PSAT_WARNINGS})
  target_compile_definitions(psat_tests PRIVATE PSAT_SOURCE_DIR="${CMAKE_SOURCE_DIR}")
  add_test(NAME unit COMMAND psat_tests)

  # Small DRAT checker (RUP + RAT) for fuzzing; uses the test-only checker.
  add_executable(psat_dratcheck tools/dratcheck.cpp)
  target_link_libraries(psat_dratcheck PRIVATE psat_lib)
  target_compile_options(psat_dratcheck PRIVATE ${PSAT_WARNINGS})
endif()
